cd /path/to/certificat_folder

# create auto-signed certificate
openssl req -x509 -nodes -days 365 -newkey rsa:2048 -keyout tls.key -out tls.crt -subj "/CN=pygeoapi.local/O=pygeoapi"
# register certiticat with kubernetes
kubectl create secret tls pygeoapi-tls --key tls.key --cert tls.crt --namespace default
